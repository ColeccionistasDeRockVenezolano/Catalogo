import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';
import path from 'node:path';
import { mkdir, writeFile } from 'node:fs/promises';
const root = process.cwd();
const require = createRequire(path.join(root, 'web/package.json'));
const { chromium, expect } = require('@playwright/test');
const { createServer } = await import(pathToFileURL(path.join(path.dirname(require.resolve('vite/package.json')), 'dist/node/index.js')).href);
const react = (await import(pathToFileURL(require.resolve('@vitejs/plugin-react')).href)).default;
const out = path.join(root, 'reports/plan-20-de-20/etapa-1/implementation-current/visual');
await mkdir(out, { recursive: true });
const sources = [
  {url:'https://example.org/archive',title:'Documento conservado de la discografía',origin:'page',contentHash:'fixture',fetchedAt:'2001-05-06T00:00:00Z',capturedAt:'2001-05-06T00:00:00Z',readAt:'2026-10-10T00:00:00Z',acquisitionMethod:'raw_archive'},
  {url:'https://example.org/historical',title:'Informe histórico sin metadatos de captura',origin:'page',contentHash:'fixture-old',fetchedAt:'2015-01-01T00:00:00Z'},
];
const report={kind:'artist',entityId:1,findings:[],areas:[],sources,limitations:[],researched:true,score:null};
const comparison={version:1,left:{kind:'artist',id:1,name:'Artista de prueba'},right:{kind:'person',id:2,name:'Persona de prueba'},createdAt:'2026-10-10T00:00:00Z',researched:true,stale:false,score:null,catalogPoints:0,webPoints:0,rows:[],conclusion:'inconclusive',explanation:'Vista controlada para comprobar las fechas de las fuentes.',evidence:sources.map(source=>({...source,quote:'Fragmento documentado de la fuente conservada.'})),limitations:[]};
comparison.left.verification=report;comparison.right.verification={...report,kind:'person',entityId:2};
const html=`<!doctype html><html lang="es"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"></head><body><main style="max-width:900px;margin:auto;padding:24px"><h1>Procedencia de las fuentes</h1><div id="root"></div></main><script type="module">
import React from 'react'; import {createRoot} from 'react-dom/client'; import {BrowserRouter} from 'react-router-dom';
import {VerificationReportDetails} from '/src/components/VerificationReportDetails.tsx';
import {EntityComparison} from '/src/components/EntityComparison.tsx';
import {OperatorProvider} from '/src/lib/OperatorContext.tsx'; import {ToastProvider} from '/src/lib/ToastContext.tsx';
import '/src/styles/global.css';
createRoot(document.getElementById('root')).render(React.createElement(BrowserRouter,{},React.createElement(ToastProvider,{},React.createElement(OperatorProvider,{},React.createElement(React.Fragment,{},React.createElement(VerificationReportDetails,{report:${JSON.stringify(report)}}),React.createElement(EntityComparison,{source:{kind:'artist',id:1},revision:1,onUpdated:()=>{}}))))));
</script></body></html>`;
let server,browser;
const results=[];
try{
  server=await createServer({root:path.join(root,'web'),configFile:false,plugins:[react(),{name:'provenance-audit',configureServer(s){s.middlewares.use(async(req,res,next)=>{if(req.url!=='/__provenance')return next();try{res.setHeader('Content-Type','text/html');res.end(await s.transformIndexHtml('/__provenance',html));}catch(error){next(error);}});}}],define:{'import.meta.env.VITE_API_BASE_URL':JSON.stringify('/api')},server:{host:'127.0.0.1',port:0}});
  await server.listen();browser=await chromium.launch({headless:true});
  for(const viewport of [{width:1440,height:1000},{width:390,height:844}]){
    const page=await browser.newPage({viewport});const errors=[];page.on('pageerror',error=>errors.push(error.message));
    await page.route('**/api/**',async route=>{
      const pathname=new URL(route.request().url()).pathname;
      if(pathname.endsWith('/auth/me'))return route.fulfill({json:{user:null}});
      if(pathname.includes('/compare/history/'))return route.fulfill({json:[{target:{kind:'person',id:2},name:'Persona de prueba',status:'done'}]});
      if(pathname.endsWith('/active-job'))return route.fulfill({json:null});
      if(pathname.endsWith('/compare'))return route.fulfill({json:comparison});
      return route.fulfill({status:500,json:{error:{message:'Unexpected fixture request'}}});
    });
    await page.goto(`http://127.0.0.1:${server.httpServer.address().port}/__provenance`);
    await page.locator('summary').filter({hasText:'Fuentes de la investigación'}).click();
    const source=page.locator('.verification-browser__source').first();
    await expect(source).toContainText('Capturada 6/5/2001');await expect(source).toContainText('Leída 10/10/2026');
    await expect(page.locator('.verification-browser__source').nth(1)).toContainText('Fecha histórica sin verificar');
    await page.getByRole('button',{name:'Comparar con otra ficha',exact:true}).click();
    await page.locator('summary').filter({hasText:'Comparaciones guardadas'}).click();
    await page.getByRole('button',{name:/Persona de prueba/}).click();
    try{await page.locator('summary').filter({hasText:'Evidencia conjunta'}).click();}catch(error){console.log(JSON.stringify({errors,body:await page.locator('body').innerText()}));throw error;}
    await expect(page.locator('.entity-comparison__evidence')).toContainText('Capturada 6/5/2001');
    await expect(page.locator('.entity-comparison__evidence')).toContainText('Captura original sin fecha verificada');
    const layout=await page.evaluate(()=>({width:innerWidth,documentWidth:document.documentElement.scrollWidth,invalidDate:document.body.innerText.includes('Invalid Date')}));
    if(errors.length||layout.documentWidth>layout.width||layout.invalidDate)throw new Error(JSON.stringify({errors,layout}));
    await page.screenshot({path:path.join(out,`provenance-${viewport.width}.png`),fullPage:true});
    results.push({viewport,layout,errors,sourceAndComparisonCaptureDatesCorrect:true,backend:'controlled read-only fixtures; no writes'});await page.close();
  }
  await writeFile(path.join(out,'browser.json'),JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(results));
}finally{await browser?.close();await server?.close();}
