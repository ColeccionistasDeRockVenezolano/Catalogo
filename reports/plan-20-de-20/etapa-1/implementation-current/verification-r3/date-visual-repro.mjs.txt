import { createRequire } from 'node:module';
import { pathToFileURL } from 'node:url';
import path from 'node:path';
import { mkdir, writeFile } from 'node:fs/promises';
const root = process.cwd();
const require = createRequire(path.join(root, 'web/package.json'));
const { chromium, expect } = require('@playwright/test');
const { createServer } = await import(pathToFileURL(path.join(path.dirname(require.resolve('vite/package.json')), 'dist/node/index.js')).href);
const react = (await import(pathToFileURL(require.resolve('@vitejs/plugin-react')).href)).default;
const out = path.join(root, 'reports/plan-20-de-20/etapa-1/implementation-current/verification-r3/date-visual');
await mkdir(out, { recursive: true });
const person={id:1,name:'Persona QA con fechas parciales',realName:null,birthCity:null,gender:null,nationality:'Venezuela',isVenezuelan:true,isDeceased:true,isDeceasedFlag:true,deathCause:null,birthDate:'1973-05',deathDate:'2020',bands:[],albumCredits:[],trackCredits:[],organizations:[],aliases:[],pictureUrl:null,biography:null,nameClass:'ok',nameClassReason:null};
const html=`<!doctype html><html lang="es"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"></head><body><main style="max-width:1100px;margin:auto;padding:24px"><div id="root"></div></main><script type="module">
import React from 'react'; import {createRoot} from 'react-dom/client'; import {BrowserRouter,Routes,Route} from 'react-router-dom';
import {PersonDetailPage} from '/src/pages/PersonDetailPage.tsx';
import {OperatorProvider} from '/src/lib/OperatorContext.tsx'; import {ToastProvider} from '/src/lib/ToastContext.tsx';
import '/src/styles/global.css';
createRoot(document.getElementById('root')).render(React.createElement(BrowserRouter,{},React.createElement(ToastProvider,{},React.createElement(OperatorProvider,{},React.createElement(Routes,{},React.createElement(Route,{path:'/personas/:id',element:React.createElement(PersonDetailPage)}))))));
</script></body></html>`;
let server,browser;
const results=[];
try{
  server=await createServer({root:path.join(root,'web'),configFile:false,plugins:[react(),{name:'provenance-audit',configureServer(s){s.middlewares.use(async(req,res,next)=>{if(req.url!=='/personas/1')return next();try{res.setHeader('Content-Type','text/html');res.end(await s.transformIndexHtml('/personas/1',html));}catch(error){next(error);}});}}],define:{'import.meta.env.VITE_API_BASE_URL':JSON.stringify('/api')},server:{host:'127.0.0.1',port:0}});
  await server.listen();browser=await chromium.launch({headless:true});
  for(const viewport of [{width:1440,height:1000},{width:390,height:844}]){
    const page=await browser.newPage({viewport});const errors=[];page.on('pageerror',error=>errors.push(error.message));
    await page.route('**/api/**',async route=>{
      const pathname=new URL(route.request().url()).pathname;
      if(pathname.endsWith('/auth/me'))return route.fulfill({json:{user:null}});
      if(pathname.endsWith('/persons/1'))return route.fulfill({json:person});
      return route.fulfill({status:500,json:{error:{message:'Unexpected fixture request'}}});
    });
    await page.goto(`http://127.0.0.1:${server.httpServer.address().port}/personas/1`);
    await expect(page.getByRole('heading',{name:person.name})).toBeVisible();
    await expect(page.locator('.entity-info')).toContainText('mayo de 1973');
    await expect(page.locator('.entity-info')).toContainText('2020');
    await expect(page.locator('.entity-info')).not.toContainText('1 de mayo');
    await expect(page.locator('.entity-info')).not.toContainText('Edad al fallecer');
    const layout=await page.evaluate(()=>({width:document.documentElement.clientWidth,documentWidth:document.documentElement.scrollWidth,invalidDate:document.body.innerText.includes('Invalid Date')}));
    if(errors.length||layout.documentWidth>layout.width||layout.invalidDate)throw new Error(JSON.stringify({errors,layout}));
    await page.screenshot({path:path.join(out,`date-precision-${viewport.width}.png`),fullPage:true});
    results.push({viewport,layout,errors,partialDateDisplayAndUnknownAgeCorrect:true,backend:'controlled read-only fixtures; no writes'});await page.close();
  }
  await writeFile(path.join(out,'browser.json'),JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(results));
}finally{await browser?.close();await server?.close();}
